def seed_everything(seed=42):
    random.seed(seed)

    torch.manual_seed(seed)

    torch.backends.cudnn.deterministic = True

def format_execution_time(seconds):
    mins, secs = divmod(seconds, 60)
    hrs, mins = divmod(mins, 60)
    if hrs > 0:
        return f"{int(hrs)}h {int(mins)}m {secs:.1f}s"
    return f"{int(mins)}m {secs:.1f}s"

def load_raman_dataset(data_type='Processed'):
    print(f"[INFO] Loading dataset variant: {data_type} from {DATA_FOLDER}...")
    healthy_path = os.path.join(DATA_FOLDER, f'{data_type} Healthy.csv')
    asymptomatic_path = os.path.join(DATA_FOLDER, f'{data_type} Asymptomatic HLB.csv')
    symptomatic_path = os.path.join(DATA_FOLDER, f'{data_type} Symptomatic HLB.csv')

    for p in [healthy_path, asymptomatic_path, symptomatic_path]:
        if not os.path.exists(p):
            raise FileNotFoundError(f"No se encontró el archivo: {p}")

    def read_spectral_csv(path):
        df = pd.read_csv(path, header=0, index_col=0).dropna()
        wavenumbers = df.columns.astype(float).values if df.columns.dtype != object else np.arange(df.shape[1])
        return df.values.astype(np.float32), wavenumbers

    h_data, wavenumbers = read_spectral_csv(healthy_path)
    a_data, _ = read_spectral_csv(asymptomatic_path)
    s_data, _ = read_spectral_csv(symptomatic_path)

    X = np.vstack([h_data, a_data, s_data])
    y = np.array([0] * len(h_data) + [1] * len(a_data) + [2] * len(s_data), dtype=np.int64)
    X_1D = X[:, np.newaxis, :]  # Shape Native 1D: (N, 1, Features)

    return X_1D, y, wavenumbers

class RamanSpectralDataset(Dataset):
    def __init__(self, X_data, y_data):
        self.X = torch.tensor(X_data, dtype=torch.float32)
        self.y = torch.tensor(y_data, dtype=torch.long)
    def __len__(self):
        return len(self.X)
    def __getitem__(self, idx):
        return self.X[idx], self.y[idx]

# 3. NATIVE 1D RESNET ARCHITECTURES
# ==========================================
class BasicBlock1D(nn.Module):
    expansion = 1
    def __init__(self, in_planes, planes, stride=1):
        super(BasicBlock1D, self).__init__()
        self.conv1 = nn.Conv1d(in_planes, planes, kernel_size=7, stride=stride, padding=3, bias=False)
        self.bn1 = nn.BatchNorm1d(planes)
        self.relu = nn.ReLU(inplace=True)
        self.conv2 = nn.Conv1d(planes, planes, kernel_size=7, stride=1, padding=3, bias=False)
        self.bn2 = nn.BatchNorm1d(planes)

        self.shortcut = nn.Sequential()
        if stride != 1 or in_planes != self.expansion * planes:
            self.shortcut = nn.Sequential(
                nn.Conv1d(in_planes, self.expansion * planes, kernel_size=1, stride=stride, bias=False),
                nn.BatchNorm1d(self.expansion * planes)
            )

    def forward(self, x):
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += self.shortcut(x)
        out = self.relu(out)
        return out

class ResNet1D(nn.Module):
    def __init__(self, block, num_blocks, num_classes=3):
        super(ResNet1D, self).__init__()
        self.in_planes = 64
        self.conv1 = nn.Conv1d(1, 64, kernel_size=15, stride=2, padding=7, bias=False)
        self.bn1 = nn.BatchNorm1d(64)
        self.relu = nn.ReLU(inplace=True)
        self.layer1 = self._make_layer(block, 64, num_blocks[0], stride=1)
        self.layer2 = self._make_layer(block, 128, num_blocks[1], stride=2)
        self.layer3 = self._make_layer(block, 256, num_blocks[2], stride=2)
        self.layer4 = self._make_layer(block, 512, num_blocks[3], stride=2)
        self.avgpool = nn.AdaptiveAvgPool1d(1)
        self.linear = nn.Linear(512 * block.expansion, num_classes)

    def _make_layer(self, block, planes, num_blocks, stride):
        strides = [stride] + [1]*(num_blocks-1)
        layers = []
        for stride in strides:
            layers.append(block(self.in_planes, planes, stride))
            self.in_planes = planes * block.expansion
        return nn.Sequential(*layers)

    def forward(self, x):
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.layer1(out)
        out = self.layer2(out)
        out = self.layer3(out)
        out = self.layer4(out)
        out = self.avgpool(out)
        out = torch.flatten(out, 1)
        out = self.linear(out)
        return out

def build_resnet18_1d(strategy='scratch'):
    model = ResNet1D(BasicBlock1D, [2, 2, 2, 2])
    if strategy == 'transfer_finetune':
        for param in model.conv1.parameters(): param.requires_grad = False
        for param in model.bn1.parameters(): param.requires_grad = False
        for param in model.layer1.parameters(): param.requires_grad = False
    return model

def build_resnet152_1d(strategy='scratch'):
    model = ResNet1D(BasicBlock1D, [3, 8, 36, 3])
    if strategy == 'transfer_finetune':
        for param in model.conv1.parameters(): param.requires_grad = False
        for param in model.bn1.parameters(): param.requires_grad = False
        for param in model.layer1.parameters(): param.requires_grad = False
    return model

# ==========================================
# 4. GRAD-CAM EXPLAINABILITY
# ==========================================
class GradCAM1D:
    def __init__(self, model, target_layer):
        self.model = model
        self.target_layer = target_layer
        self.gradients = None
        self.activations = None
        
        self.target_layer.register_forward_hook(self.save_activation)
        self.target_layer.register_full_backward_hook(self.save_gradient)
        
    def save_activation(self, module, input, output):
        self.activations = output
        
    def save_gradient(self, module, grad_input, grad_output):
        self.gradients = grad_output[0]
        
    def generate_cam(self, input_tensor, target_class):
        self.model.eval()
        output = self.model(input_tensor)
        self.model.zero_grad()
        loss = output[0, target_class]
        loss.backward()
        
        gradients = self.gradients[0].cpu().data.numpy()
        activations = self.activations[0].cpu().data.numpy()
        weights = np.mean(gradients, axis=1)
        
        cam = np.zeros(activations.shape[1], dtype=np.float32)
        for i, w in enumerate(weights):
            cam += w * activations[i]
            
        cam = np.maximum(cam, 0)
        cam = (cam - np.min(cam)) / (np.max(cam) - np.min(cam) + 1e-8)
        return np.interp(np.linspace(0, 1, input_tensor.shape[-1]), np.linspace(0, 1, len(cam)), cam)

def plot_gradcam_spectrum(wavenumbers, spectrum, cam, title, save_path):
    plt.figure(figsize=(10, 4))
    plt.plot(wavenumbers, spectrum, label='Mean Spectrum', color='black', alpha=0.7)
    plt.scatter(wavenumbers, spectrum, c=cam, cmap='jet', s=15, zorder=3)
    plt.colorbar(label='Grad-CAM Importance (0 to 1)')
    plt.xlabel('Raman Shift (cm⁻¹)')
    plt.ylabel('Intensity')
    plt.title(title)
    plt.tight_layout()
    plt.savefig(save_path, dpi=300)
    plt.close()

# 5. TRAINING & EVALUATION PIPELINE (5-FOLD)
# ==========================================
def train_and_evaluate_experiment(exp_name, build_fn, strategy, X_data, y_data, wavenumbers):
    exp_start_time = time.time()
    print(f"\n" + "="*75)
    print(f"RUNNING EXPERIMENT: {exp_name}")
    print(f"="*75)

    counts = np.bincount(y_data)
    class_weights = torch.tensor(len(y_data) / (len(counts) * counts), dtype=torch.float32).to(device)

    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=SEED)
    fold_metrics = []
    confusion_matrices = []
    best_overall_model = None
    best_overall_val_loss = float('inf')

    for fold, (train_idx, val_idx) in enumerate(skf.split(X_data, y_data)):
        train_ds = RamanSpectralDataset(X_data[train_idx], y_data[train_idx])
        val_ds = RamanSpectralDataset(X_data[val_idx], y_data[val_idx])
        
        train_loader = DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True)
        val_loader = DataLoader(val_ds, batch_size=BATCH_SIZE, shuffle=False)
        
        model = build_fn(strategy=strategy).to(device)
        criterion = nn.CrossEntropyLoss(weight=class_weights)
        
        lr = 1e-4 if strategy == 'transfer_finetune' else 1e-3
        optimizer = optim.Adam(filter(lambda p: p.requires_grad, model.parameters()), lr=lr, weight_decay=1e-4)
        scheduler = optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode='min', factor=0.5, patience=4)
        
        best_val_loss = float('inf')
        patience_counter = 0
        best_weights = None
        
        for epoch in range(1, MAX_EPOCHS + 1):
            model.train()
            for Xb, yb in train_loader:
                Xb, yb = Xb.to(device), yb.to(device)
                optimizer.zero_grad()
                out = model(Xb)
                loss = criterion(out, yb)
                loss.backward()
                optimizer.step()
                
            model.eval()
            val_loss = 0.0
            with torch.no_grad():
                for Xb, yb in val_loader:
                    Xb, yb = Xb.to(device), yb.to(device)
                    out = model(Xb)
                    loss = criterion(out, yb)
                    val_loss += loss.item() * len(yb)
                    
            val_loss /= len(val_ds)
            scheduler.step(val_loss)
            
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                patience_counter = 0
                best_weights = model.state_dict().copy()
            else:
                patience_counter += 1
                
            if patience_counter >= PATIENCE:
                break
                
        model.load_state_dict(best_weights)
        if best_val_loss < best_overall_val_loss:
            best_overall_val_loss = best_val_loss
            best_overall_model = model

        # Evaluation per Fold
        model.eval()
        fold_preds, fold_probs, fold_targets = [], [], []
        with torch.no_grad():
            for Xb, yb in val_loader:
                Xb = Xb.to(device)
                outputs = model(Xb)
                probs = torch.softmax(outputs, dim=1)
                fold_probs.append(probs.cpu().numpy())
                fold_preds.append(torch.argmax(probs, dim=1).cpu().numpy())
                fold_targets.append(yb.numpy())
                
        f_preds = np.concatenate(fold_preds)
        f_targets = np.concatenate(fold_targets)
        f_probs = np.concatenate(fold_probs)
        
        prec, rec, f1, _ = precision_recall_fscore_support(f_targets, f_preds, average='macro')
        bal_acc = balanced_accuracy_score(f_targets, f_preds)
        
        cm = confusion_matrix(f_targets, f_preds)
        confusion_matrices.append(cm)
        specificities = []
        for i in range(3):
            tp = cm[i, i]
            fn = np.sum(cm[i, :]) - tp
            fp = np.sum(cm[:, i]) - tp
            tn = np.sum(cm) - (tp + fn + fp)
            specificities.append(tn / (tn + fp + 1e-8))
        spec_macro = np.mean(specificities)
        
        auc_macro = roc_auc_score(f_targets, f_probs, multi_class='ovr', average='macro')
        y_onehot = pd.get_dummies(f_targets).values
        auc_micro = roc_auc_score(y_onehot, f_probs, average='micro')
        
        fold_metrics.append([prec, rec, spec_macro, bal_acc, f1, auc_macro, auc_micro])

    metrics_arr = np.array(fold_metrics)
    means = np.mean(metrics_arr, axis=0)
    stds = np.std(metrics_arr, axis=0, ddof=1)
    
    # 90% Confidence Interval (z = 1.645 for N=5)
    ci_90 = 1.645 * (stds / np.sqrt(5))

    # Save Mean Confusion Matrix
    cm_mean, cm_std = np.mean(confusion_matrices, axis=0), np.std(confusion_matrices, axis=0, ddof=1)
    plt.figure(figsize=(6, 5))
    annot = np.empty_like(cm_mean, dtype=object)
    for i in range(3):
        for j in range(3):
            annot[i, j] = f"{cm_mean[i, j]:.1f} ± {cm_std[i, j]:.1f}"
            
    sns.heatmap(cm_mean, annot=annot, fmt='', cmap='Blues', xticklabels=CLASS_NAMES, yticklabels=CLASS_NAMES)
    plt.title(f'CM: {exp_name}\n[Mean ± Std (5-Fold CV)]')
    plt.xlabel('Predicted')
    plt.ylabel('Actual')
    plt.tight_layout()
    plt.savefig(os.path.join(RESULTS_FOLDER, f'CM_{exp_name}.png'), dpi=300)
    plt.close()

    # Generate Grad-CAM Visualizations
    cam_extractor = GradCAM1D(best_overall_model, best_overall_model.layer4)
    for cls_idx, cls_name in enumerate(CLASS_NAMES):
        class_samples = X_data[y_data == cls_idx]
        mean_spectrum = np.mean(class_samples, axis=0)
        input_tensor = torch.tensor(mean_spectrum, dtype=torch.float32).unsqueeze(0).to(device)
        cam = cam_extractor.generate_cam(input_tensor, cls_idx)
        plot_gradcam_spectrum(
            wavenumbers, mean_spectrum[0], cam,
            title=f'Grad-CAM: {exp_name} - Class: {cls_name}',
            save_path=os.path.join(GRADCAM_FOLDER, f'GradCAM_{exp_name}_{cls_name}.png')
        )

    exp_elapsed = time.time() - exp_start_time
    print(f"Execution Time ({exp_name}): {format_execution_time(exp_elapsed)}")

    return {
        'Approach': exp_name,
        'Precision': f"{means[0]:.4f} ± {stds[0]:.4f} (90% CI: [{means[0]-ci_90[0]:.4f}, {means[0]+ci_90[0]:.4f}])",
        'Recall': f"{means[1]:.4f} ± {stds[1]:.4f} (90% CI: [{means[1]-ci_90[1]:.4f}, {means[1]+ci_90[1]:.4f}])",
        'Specificity': f"{means[2]:.4f} ± {stds[2]:.4f}",
        'Balanced Acc': f"{means[3]:.4f} ± {stds[3]:.4f}",
        'F1': f"{means[4]:.4f} ± {stds[4]:.4f} (90% CI: [{means[4]-ci_90[4]:.4f}, {means[4]+ci_90[4]:.4f}])",
        'AUC Macro': f"{means[5]:.4f} ± {stds[5]:.4f}",
        'AUC Micro': f"{means[6]:.4f} ± {stds[6]:.4f}",
        'Execution Time': format_execution_time(exp_elapsed)
    }

# ==========================================
# 6. MAIN EXECUTION PIPELINE
# ==========================================
def main():
    X_proc, y_proc, wave_proc = load_raman_dataset('Processed')
    X_raw, y_raw, wave_raw = load_raman_dataset('Raw')

    experiments = [
        # ResNet-18
        ('Based on ResNet-18 (Raw)', build_resnet18_1d, 'scratch', X_raw, y_raw, wave_raw),
        ('Based on ResNet-18 (Processed)', build_resnet18_1d, 'scratch', X_proc, y_proc, wave_proc),
        ('ResNet-18 Transfer Learning (Raw)', build_resnet18_1d, 'transfer_finetune', X_raw, y_raw, wave_raw),
        ('ResNet-18 Transfer Learning (Processed)', build_resnet18_1d, 'transfer_finetune', X_proc, y_proc, wave_proc),
        
        # ResNet-152
        ('Based on ResNet-152 (Raw)', build_resnet152_1d, 'scratch', X_raw, y_raw, wave_raw),
        ('Based on ResNet-152 (Processed)', build_resnet152_1d, 'scratch', X_proc, y_proc, wave_proc),
        ('ResNet-152 Transfer Learning (Raw)', build_resnet152_1d, 'transfer_finetune', X_raw, y_raw, wave_raw),
        ('ResNet-152 Transfer Learning (Processed)', build_resnet152_1d, 'transfer_finetune', X_proc, y_proc, wave_proc),
    ]

    summary_results = []
    for exp_name, build_fn, strategy, X_d, y_d, waves in experiments:
        res = train_and_evaluate_experiment(exp_name, build_fn, strategy, X_d, y_d, waves)
        summary_results.append(res)

    total_elapsed = time.time() - GLOBAL_START_TIME

    df_summary = pd.DataFrame(summary_results)
    df_summary.to_csv(os.path.join(RESULTS_FOLDER, 'Metrics_ResNet_1D_Publication.csv'), index=False)

    print("\n" + "="*120)
    print(" FINAL PUBLICATION SUMMARY TABLE (Mean ± Std, 90% CI, Grad-CAM & Execution Times)")
    print("="*120)
    print(df_summary.to_string(index=False))
    print("="*120)
    print(f"TOTAL SUITE EXECUTION TIME: {format_execution_time(total_elapsed)}")
    print("="*120)

if __name__ == '__main__':
    main()
